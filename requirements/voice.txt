# onnxruntime-gpu's CUDA provider links the full CUDA 12 runtime, so cuDNN/cuBLAS
# alone are NOT enough — cudart, cuFFT and cuRAND are also required. These wheels
# supply every DLL, so no system CUDA toolkit is needed. Machines without a CUDA
# GPU install them but never load them (auto-falls back to CPUExecutionProvider).
# <1.27 pin: 1.27 fails to load the CUDA provider against this cu12 wheel set.
# Keep in sync with _ORT_GPU_SPEC in setup.py.
onnxruntime-gpu<1.27 ; platform_system != "Darwin"
onnxruntime ; platform_system == "Darwin"
nvidia-cuda-runtime-cu12>=12.0,<13 ; platform_system == "Windows"
nvidia-cudnn-cu12>=9.0,<10 ; platform_system == "Windows"
nvidia-cublas-cu12>=12.0,<13 ; platform_system == "Windows"
nvidia-cufft-cu12>=11.0,<12 ; platform_system == "Windows"
nvidia-curand-cu12>=10.0,<11 ; platform_system == "Windows"
